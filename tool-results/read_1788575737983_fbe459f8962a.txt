   393→   393→          paymentAmount: payAmount,
   394→   394→          customerName: isUtang ? customerName.trim() : undefined,
   395→   395→        }),
   396→   396→      });
   397→   397→
   398→   398→      if (res.error) { toast.error(res.error); return; }

   411→   411→    }
   412→   412→  };

   452→   452→            </div>
   453→   453→            {/* Manual barcode input below search */}
   454→   454→            <div className="mt-1.5">
   455→   455→              <div className="flex gap-2">
   456→   456→                <Input
   457→   457→                  placeholder="Type barcode number and press Enter…"
   458→   458→                  className="h-9 text-xs"
   459→   459→                  value={scannerBusy ? '' : undefined}

   490→   490→
