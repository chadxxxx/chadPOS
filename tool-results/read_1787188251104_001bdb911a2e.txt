   102→   102→  const [showScanner, setShowScanner] = useState(false);

   188→   188→  }, []);
   189→   189→
   190→   190→  const searchProducts = useCallback(async (query: string) => {

   200→   200→        const d = res.data as any;
   201→   201→        const products: Product[] = d.data || d;
   202→   202→        setSearchResults(products);
   203→   203→        setShowResults(true);
   204→   204→      }

   207→   207→    } finally {
   208→   208→      setSearching(false);

   221→   221→        setShowResults(true);

   224→   224→      // silent
   225→   225→    } finally {
   226→   226→      setSearching(false);
   227→   227→      setBrowsing(false);

   274→   274→            name: product.name,

   359→   359→            productId: i.productId,
   360→   360→            name: i.name,
   361→   361→            barcode: i.barcode,
   362→   362→            price: i.price,

   375→   375→      if (res.data) {

   384→   384→    } finally {

   386→   386→    }
   387→   387→  };
   388→   388→
   389→   389→  return (

   391→   391→      <div className="grid gap-4 lg:grid-cols-[1fr_380px]">
   392→   392→        {/* Left: Search + Cart */}
   393→   393→        <div className="space-y-4">
   394→   394→          {/* Search bar with manual barcode input */}
   395→   395→          <div className="relative" ref={searchRef}>
   396→   396→            <div className="flex gap-2">
   397→   397→              <div className="relative flex-1">
   398→   398→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />

   400→   400→                  ref={searchInputRef}

   406→   406→                  autoFocus
   407→   407→                />
   408→   408→              </div>
   409→   409→              <Button
   410→   410→                variant="outline"
   411→   411→                className="h-11 px-3 shrink-0 text-xs"
   412→   412→                onClick={browseProducts}

   417→   417→              </Button>
   418→   418→              <Button

   420→   420→                size="icon"
   421→   421→                className="h-11 w-11 shrink-0"

   423→   423→                title="Camera Barcode Scanner"
   424→   424→              >
   425→   425→                <ScanBarcode className="h-4 w-4" />
   426→   426→              </Button>
   427→   427→            </div>
   428→   428→            {/* Manual barcode input below search */}
   429→   429→            <div className="mt-1.5">
   430→   430→              <div className="flex gap-2">
   431→   431→                <Input

   437→   437→                    if (e.key === 'Enter') {
   438→   438→                      const val = (e.target as HTMLInputElement).value.trim();
   439→   439→                      if (val.length >= 3) {
   440→   440→                        e.preventDefault();
   441→   441→                        (e.target as HTMLInputElement).value = '';
   442→   442→                        handleBarcodeLookup(val);

   444→   444→                    }

   446→   446→                />
   447→   447→                <Button

   449→   449→                  size="sm"

   454→   454→                    const val = input?.value?.trim();

   457→   457→                      handleBarcodeLookup(val);
   458→   458→                    }
   459→   459→                  }}
   460→   460→                >
   461→   461→                  Look Up
   462→   462→                </Button>
   463→   463→              </div>
   464→   464→            </div>
   465→   465→
   466→   466→            {/* Search results dropdown */}
   467→   467→            {showResults && searchResults.length > 0 && (

   474→   474→                  >
